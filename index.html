<script>

let skor = 0;

let nyawa = 3;

let waktu = 15;

let jawabanBenar = 0;

let intervalTimer;


// Memulai game pertama kali

mulaiGame();


function mulaiGame() {

skor = 0;

nyawa = 3;

document.getElementById('play-screen').style.display = 'block';

document.getElementById('game-over-screen').style.display = 'none';

perbaruiStatus();

buatSoalBaru();

}


function buatSoalBaru() {

waktu = 15;

document.getElementById('timer').innerText = `Waktu: ${waktu}s`;

clearInterval(intervalTimer);

intervalTimer = setInterval(() => {

waktu--;

document.getElementById('timer').innerText = `Waktu: ${waktu}s`;

if (waktu <= 0) {

clearInterval(intervalTimer);

kurangiNyawa();

}

}, 1000);


const listOperator = ['+', '-', '×', '÷'];

let operator = listOperator[Math.floor(Math.random() * listOperator.length)];


let angka1, angka2;


if (operator === '+') {

angka1 = Math.floor(Math.random() * 30) + 1;

angka2 = Math.floor(Math.random() * 30) + 1;

jawabanBenar = angka1 + angka2;

}

else if (operator === '-') {

angka1 = Math.floor(Math.random() * 30) + 1;

angka2 = Math.floor(Math.random() * 30) + 1;

if (angka1 < angka2) {

let temp = angka1;

angka1 = angka2;

angka2 = temp;

}

jawabanBenar = angka1 - angka2;

}

else if (operator === '×') {

angka1 = Math.floor(Math.random() * 10) + 1;

angka2 = Math.floor(Math.random() * 10) + 1;

jawabanBenar = angka1 * angka2;

}

else if (operator === '÷') {

angka2 = Math.floor(Math.random() * 10) + 1;

jawabanBenar = Math.floor(Math.random() * 10) + 1;

angka1 = angka2 * jawabanBenar;

}


// ========================================== -->

<!-- 2. MEMBERITAHU POKI GAMEPLAY DIMULAI -->

<!-- ========================================== -->

if (typeof PokiSDK !== 'undefined') {

PokiSDK.gameplayStart();

}


document.getElementById('question').innerText = `${angka1} ${operator} ${angka2} = ?`;


let pilihan = [jawabanBenar];

while (pilihan.length < 4) {

let jawabanSalah = jawabanBenar + (Math.floor(Math.random() * 9) - 4);

if (jawabanSalah !== jawabanBenar && !pilihan.includes(jawabanSalah) && jawabanSalah >= 0) {

pilihan.push(jawabanSalah);

}

}


pilihan.sort(() => Math.random() - 0.5);


let tombols = document.querySelectorAll('.btn-option');

for (let i = 0; i < tombols.length; i++) {

tombols[i].innerText = pilihan[i];

}

}


function periksaJawaban(tombol) {

let jawabanPemain = parseInt(tombol.innerText);

if (jawabanPemain === jawabanBenar) {

skor += 10;

perbaruiStatus();

buatSoalBaru();

} else {

kurangiNyawa();

}

}


function kurangiNyawa() {

nyawa--;

perbaruiStatus();

if (nyawa <= 0) {

clearInterval(intervalTimer);

gameOver();

} else {

buatSoalBaru();

}

}


function perbaruiStatus() {

document.getElementById('score').innerText = `Skor: ${skor}`;

let hati = '';

for (let i = 0; i < nyawa; i++) {

hati += '❤️';

}

document.getElementById('lives').innerText = `Nyawa: ${hati}`;

}


function gameOver() {

// ========================================== -->

<!-- 3. MEMBERITAHU POKI GAMEPLAY BERHENTI -->

<!-- ========================================== -->

if (typeof PokiSDK !== 'undefined') {

PokiSDK.gameplayStop();

}


document.getElementById('play-screen').style.display = 'none';

document.getElementById('game-over-screen').style.display = 'block';

document.getElementById('final-score').innerText = skor;

}
